Example: Fabric Mod Structure

// Hero Studios -- modular Fabric mod architecture
public class HeroClient implements ClientModInitializer {
  public static final String MOD_ID = "hero-client";
  public static final Logger LOG = LoggerFactory.getLogger(MOD_ID);

  @Override
  public void onInitializeClient() {
    LOG.info("Loading {} v{}", MOD_ID, BuildConfig.VERSION);
    ConfigManager.load();
    EventManager.register(new PerformanceModule());
    EventManager.register(new EcosystemBridge());
    EventManager.register(new UIScheduler());
  }
}
